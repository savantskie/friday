@app.api_route("/mcp", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy_mcp_root(
    request: Request,
    current_user: dict = Depends(get_current_user)
):
    """
    Proxy MCP root endpoint to MCPO backend

    This handles requests to /mcp (without additional path)
    which is the base endpoint for MCP protocol initialization.
    """
    target_url = MCPO_BACKEND

    # Get request body
    body = await request.body()

    # Prepare headers
    headers = dict(request.headers)
    headers.pop('authorization', None)
    headers.pop('Authorization', None)
    headers['Authorization'] = f'Bearer {MCPO_BEARER_TOKEN}'
    headers.pop('host', None)
    headers.pop('Host', None)

    logger.info(f"Proxying {request.method} request to {target_url} (MCP root)")

    try:
        async with aiohttp.ClientSession() as session:
            async with session.request(
                method=request.method,
                url=target_url,
                headers=headers,
                data=body,
                params=request.query_params
            ) as resp:
                content = await resp.read()
                response_headers = dict(resp.headers)
                response_headers.pop('transfer-encoding', None)
                response_headers.pop('Transfer-Encoding', None)

                return Response(
                    content=content,
                    status_code=resp.status,
                    headers=response_headers
                )
    except Exception as e:
        logger.error(f"Error proxying request to MCPO: {e}")
        raise HTTPException(status_code=502, detail=f"Error communicating with backend: {str(e)}")

async def init_token_database():
    """Initialize SQLite database for token storage"""
    async with aiosqlite.connect(TOKEN_DB_PATH) as db:
        await db.execute("""
            CREATE TABLE IF NOT EXISTS authorization_codes (
                code TEXT PRIMARY KEY,
                client_id TEXT NOT NULL,
                redirect_uri TEXT,
                state TEXT,
                scope TEXT,
                created_at REAL NOT NULL,
                expires_at REAL NOT NULL
            )
        """)
        await db.execute("""
            CREATE TABLE IF NOT EXISTS access_tokens (
                token TEXT PRIMARY KEY,
                client_id TEXT NOT NULL,
                created_at REAL NOT NULL,
                expires_at REAL NOT NULL
            )
        """)
        await db.execute("""
            CREATE TABLE IF NOT EXISTS refresh_tokens (
                token TEXT PRIMARY KEY,
                client_id TEXT NOT NULL,
                created_at REAL NOT NULL,
                expires_at REAL NOT NULL
            )
        """)

# Plan: Fix Claude.ai Web MCP Connection via Header-Based Routing

## Critical Discovery

OAuth proxy logs reveal that **claude.ai web strips the path from the server URL**. When configured with `https://fridayonline.bounceme.net/mcp`, claude.ai actually requests:
- `POST / HTTP/1.1` (root path)
- `GET / HTTP/1.1` (root path)
- `POST /%2A HTTP/1.1` (wildcard path `/*`)
- `GET /%2A HTTP/1.1` (wildcard path `/*`)

The `/mcp` path in the server URL configuration is **completely ignored**.

## The Problem

- Caddy routes root `/` to OpenWebUI (port 3000)
- Claude.ai requests hit OpenWebUI instead of OAuth proxy (port 8888)
- MCP connection fails with 404 errors
- **User constraint**: Base URL MUST remain for OpenWebUI (cannot afford subdomain)

## The Solution: Header-Based Routing

**Key Insight**: Claude.ai sends `Authorization: Bearer <token>` on ALL MCP requests. Browser requests to OpenWebUI do NOT send this header. We can use this to differentiate traffic.

**Strategy**:
1. Caddy matches requests with `Authorization` header → route to OAuth proxy
2. Requests without `Authorization` header → route to OpenWebUI (unchanged)
3. OAuth proxy validates token and proxies to MCPO
4. Works for both `/` and `/*` paths that claude.ai uses

## Implementation Steps

### Step 1: Update Caddy Configuration

**File**: `/media/nate/Friday/Friday/caddy/Caddyfile`

**Change** (around line 125, BEFORE the default OpenWebUI reverse_proxy):

**OLD**:

    }

**NEW**:

# Handle root path / (claude.ai hits this)
@app.api_route("/", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy_root(
    request: Request,
    current_user: dict = Depends(get_current_user)
):
    """Proxy root path requests to MCPO backend"""
    target_url = MCPO_BACKEND
    body = await request.body()

    headers = dict(request.headers)
    headers.pop('authorization', None)
    headers.pop('Authorization', None)
    headers['Authorization'] = f'Bearer {MCPO_BEARER_TOKEN}'
    headers.pop('host', None)
    headers.pop('Host', None)

    logger.info(f"Proxying {request.method} to {target_url} (root path)")

    try:
        async with aiohttp.ClientSession() as session:
            async with session.request(
                method=request.method,
                url=target_url,
                headers=headers,
                data=body,
                params=request.query_params
            ) as resp:
                content = await resp.read()
                response_headers = dict(resp.headers)
                response_headers.pop('transfer-encoding', None)
                response_headers.pop('Transfer-Encoding', None)
                return Response(
                    content=content,
                    status_code=resp.status,
                    headers=response_headers
                )
    except Exception as e:
        logger.error(f"Error proxying to MCPO: {e}")
        raise HTTPException(status_code=502, detail=f"Error: {str(e)}")


# Handle wildcard path /* (claude.ai also hits /%2A)

# Plan: Fix Claude.ai Web MCP Connection via JWT Token-Based Routing

## Critical Discovery

OAuth proxy logs reveal that **claude.ai web strips the path from the server URL**. When configured with `https://fridayonline.bounceme.net/mcp`, claude.ai actually requests:
- `POST / HTTP/1.1` (root path)
- `GET / HTTP/1.1` (root path)
- `POST /%2A HTTP/1.1` (wildcard path `/*`)
- `GET /%2A HTTP/1.1` (wildcard path `/*`)

The `/mcp` path in the server URL configuration is **completely ignored**.

## The Problem

- Caddy routes root `/` to OpenWebUI (port 3000)
- Claude.ai requests hit OpenWebUI instead of OAuth proxy (port 8888)
- MCP connection fails with 404 errors
- **User constraint**: Base URL MUST remain for OpenWebUI (cannot afford subdomain)

## The Solution: JWT Token-Based Routing (USER'S EXCELLENT INSIGHT!)

**Key Insight from User**: Claude.ai sends **JWT tokens** (start with `eyJ`), while OpenWebUI sends a **specific MCPO bearer token**. We can differentiate based on token format!

**Token Formats**:
- **Claude.ai OAuth**: `Authorization: Bearer eyJ...` (JWT from OAuth flow - always starts with `eyJ`)
- **OpenWebUI MCPO API**: `Authorization: Bearer 051407adocker pull ghcr.io/open-webui/open-webui:mainB1982c103008d!` (user's specific long key)
- **Browser (OpenWebUI UI)**: No Authorization header on page loads

**Strategy**:
1. Caddy matcher checks if Authorization header value starts with `Bearer eyJ` (JWT format)
2. JWT tokens (claude.ai) → route to OAuth proxy (port 8888)
3. Non-JWT or no token (OpenWebUI/browser) → route to OpenWebUI (port 3000)
4. **This is MORE precise than just checking for header presence - it won't interfere with OpenWebUI's MCPO calls!**

- Only claude.ai JWT requests match the pattern → routed to OAuth proxy
- Everything else (OpenWebUI UI + MCPO API) → routed to OpenWebUI
- No interference with OpenWebUI's existing MCPO functionality!

# Handle root path / (claude.ai hits this)
@app.api_route("/", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy_root(
    request: Request,
    current_user: dict = Depends(get_current_user)
):
    """Proxy root path requests to MCPO backend"""
    target_url = MCPO_BACKEND
    body = await request.body()

    headers = dict(request.headers)
    headers.pop('authorization', None)
    headers.pop('Authorization', None)
    headers['Authorization'] = f'Bearer {MCPO_BEARER_TOKEN}'
    headers.pop('host', None)
    headers.pop('Host', None)

    logger.info(f"Proxying {request.method} to {target_url} (root path)")

    try:
        async with aiohttp.ClientSession() as session:
            async with session.request(
                method=request.method,
                url=target_url,
                headers=headers,
                data=body,
                params=request.query_params
            ) as resp:
                content = await resp.read()
                response_headers = dict(resp.headers)
                response_headers.pop('transfer-encoding', None)
                response_headers.pop('Transfer-Encoding', None)
                return Response(
                    content=content,
                    status_code=resp.status,
                    headers=response_headers
                )
    except Exception as e:
        logger.error(f"Error proxying to MCPO: {e}")
        raise HTTPException(status_code=502, detail=f"Error: {str(e)}")

# Handle root path / (claude.ai hits this)
@app.api_route("/", methods=["GET", "POST", "PUT", "DELETE", "PATCH"])
async def proxy_root(
    request: Request,
    current_user: dict = Depends(get_current_user)
):
    """Proxy root path requests to MCPO backend"""
    target_url = MCPO_BACKEND
    body = await request.body()

    headers = dict(request.headers)
    headers.pop('authorization', None)
    headers.pop('Authorization', None)
    headers['Authorization'] = f'Bearer {MCPO_BEARER_TOKEN}'
    headers.pop('host', None)
    headers.pop('Host', None)

    logger.info(f"Proxying {request.method} to {target_url} (root path)")

    try:
        async with aiohttp.ClientSession() as session:
            async with session.request(
                method=request.method,
                url=target_url,
                headers=headers,
                data=body,
                params=request.query_params
            ) as resp:
                content = await resp.read()
                response_headers = dict(resp.headers)
                response_headers.pop('transfer-encoding', None)
                response_headers.pop('Transfer-Encoding', None)
                return Response(
                    content=content,
                    status_code=resp.status,
                    headers=response_headers
                )
    except Exception as e:
        logger.error(f"Error proxying to MCPO: {e}")
        raise HTTPException(status_code=502, detail=f"Error: {str(e)}")
